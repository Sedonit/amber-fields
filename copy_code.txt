    function normalizeAdmins(text) {

    function getLockUntil() {
        const raw = parseInt(store.get(LOCK_KEY) || '0', 10);
        return isNaN(raw) ? 0 : raw;
    }

    function isLocked() {
        return Date.now() < getLockUntil();
    }

    function submitPassword() {
        if (isLocked()) {
            startLockTicker();
            return;
        }

        const loginInput = gate.querySelector('input[name="admin-login"]');
        const passInput = gate.querySelector('input[name="admin-pass"]');
        const dateInput = gate.querySelector('input[name="admin-date"]');
        const err = gate.querySelector('.admin-error');

        const loginValue = loginInput.value.trim();
        const passValue = passInput.value.trim();
        const dateValue = dateInput.value.trim();

        // Строгое посимвольное совпадение — регистр тоже важен
        const okLogin = loginValue === EXPECTED.login;
        const okPass = passValue === EXPECTED.password;
        const okDate = dateValue === EXPECTED.date;

        if (okLogin && okPass && okDate) {
            loginInput.value = '';
            passInput.value = '';
            dateInput.value = '';
            err.textContent = '';
            stopLockTicker();
            log('данные верные');
            grantAccess('форма входа');
            return;
        }

        const fails = (parseInt(store.get(FAIL_KEY) || '0', 10) || 0) + 1;

        err.textContent = 'НЕВЕРНЫЕ ДАННЫЕ';
        passInput.value = '';
        dateInput.value = '';
        (okLogin ? passInput : loginInput).focus();

        const panel = gate.querySelector('.admin-panel');
        panel.classList.remove('admin-shake');
        void panel.offsetWidth;
        panel.classList.add('admin-shake');

        log('неверный пароль, попытка', fails);

        if (fails >= MAX_FAILS) {
            const until = Date.now() + LOCK_MS;
            store.set(LOCK_KEY, String(until));
            store.set(FAIL_KEY, '0');
            // «Закрыть на 30 секунд»
            if (closeTimer) clearTimeout(closeTimer);
            closeTimer = setTimeout(function () {
                closeTimer = null;
                close();
            }, 900);
            return;
        }

        store.set(FAIL_KEY, String(fails));
    }

    function startLockTicker() {
        stopLockTicker();

        const tick = function () {
            const left = Math.ceil((getLockUntil() - Date.now()) / 1000);

            if (left <= 0) {
                stopLockTicker();
                renderLock(0);
                return;
            }

            renderLock(left);
        };

        tick();
        lockTicker = setInterval(tick, 250);
    }

    function stopLockTicker() {
        if (lockTicker) {
            clearInterval(lockTicker);
            lockTicker = null;
        }
    }

    function renderLock(left) {
        const inputs = gate.querySelectorAll('.admin-input');
        const submit = gate.querySelector('button[type="submit"]');
        const err = gate.querySelector('.admin-error');
        const note = gate.querySelector('.admin-lock-note');
        const locked = left > 0;

        inputs.forEach(function (el) { el.disabled = locked; });
        submit.disabled = locked;
        gate.classList.toggle('is-locked', locked);

        if (locked) {
            err.textContent = 'ЗАБЛОКИРОВАНО: ' + left + ' С';
            note.textContent = 'Слишком много попыток. Подождите.';
            note.classList.remove('admin-hidden');
        } else {
            if (err.textContent.indexOf('ЗАБЛОКИРОВАНО') === 0) err.textContent = '';
            note.textContent = '';
            note.classList.add('admin-hidden');
        }
    }

    // ---- ОТЛОЖЕННЫЙ ЗАПРОС ДАННЫХ ---------------------------

    // ---- ПЕРЕХОДЫ -------------------------------------------
    // Останавливаем интро, иначе его async-сценарий сам сменит экран позже
    function stopIntro() {

        intro.finished = true;
        intro.isTyping = false;

            clearInterval(intro.currentInterval);
            intro.currentInterval = null;
        }
        if (typeof intro.removeClickHandler === 'function') intro.removeClickHandler();
    }

    function jumpTo(target) {
        const screen = screenMod();
        const loader = loaderMod();
        const intro = introMod();

        if (!target) return;

        // Убираем заставку техработ перед переходом на любой другой экран
        if (loader && typeof loader.hideMaintenance === 'function') loader.hideMaintenance();

        if (target === 'intro') {
            if (!screen || !intro) return flash('INTRO НЕДОСТУПНО');
            stopIntro();
            close();
            screen.showScreen('intro-screen');
            intro.init();
            log('jump intro');
            return;
        }

        stopIntro();

        switch (target) {
            case 'loader':
                if (!document.getElementById('loader-screen')) {
                    return flash('ЛОАДЕР УДАЛЁН ПОСЛЕ СТАРТА · ПЕРЕЗАГРУЗИТЕ');
                }
                if (!screen) return flash('SCREENMANAGER НЕ НАЙДЕН');
                close();
                screen.showScreen('loader-screen');
                break;

            case 'menu':
                if (!screen) return flash('SCREENMANAGER НЕ НАЙДЕН');
                close();
                screen.showScreen('main-menu');
                if (typeof MainMenu !== 'undefined') MainMenu.refreshPlayButton();
                break;

            case 'game':
                if (!screen) return flash('SCREENMANAGER НЕ НАЙДЕН');
                close();
                screen.showScreen('game-screen');
                break;

            case 'credits':
                if (!modalMod() || typeof Modal.openCredits !== 'function') {
                    return flash('CREDITS НЕДОСТУПНЫ');
                }
                close();
                Modal.openCredits();

                if (!modalMod() || typeof Modal.openHelp !== 'function') {
                    return flash('HELP НЕДОСТУПЕН');
                }
                close();
                Modal.openHelp();
                break;

            case 'maintenance':
                if (!loader || typeof loader.showMaintenance !== 'function') {
                    return flash('LOADER НЕ НАЙДЕН');
                }
                if (!document.getElementById('loader-maintenance')) {
                    return flash('ЭКРАН ТЕХРАБОТ УДАЛЁН · ПЕРЕЗАГРУЗИТЕ');

                loader.showMaintenance();
                break;

        log('jump', target);
    }

    // ---- ИНФОРМАЦИЯ -----------------------------------------
    function collectStorage() {
        const full = {};
        const preview = {};
        let bytes = 0;
        let count = 0;

        try {
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                const value = localStorage.getItem(key);
                const val = value == null ? '' : value;

                full[key] = val;
                preview[key] = val.length > 100 ? val.slice(0, 100) + '…' : val;
                bytes += (key.length + val.length) * 2; // UTF-16
                count++;
            }
        } catch (e) {
            full.__error = 'localStorage недоступен';
            preview.__error = 'localStorage недоступен';
        }

        return {
            full: full,
            preview: preview,
            count: count,
            kb: (bytes / 1024).toFixed(2)
        };
    }

    function syncInfo() {
        if (!stage) return;

        const screenLine = stage.querySelector('[data-info-screen]');
        const jsonEl = stage.querySelector('[data-info-json]');
        const metaEl = stage.querySelector('[data-info-meta]');

        if (screenLine) {
            const active = document.querySelector('.screen.active');
            let text = active ? active.id : '—';

            const modal = modalMod();
            if (modal) {
                const layers = [];
                if (modal.activeModal) layers.push(modal.activeModal);
                if (modal.creditsOpen) layers.push('credits');
                if (modal.helpOpen) layers.push('help');
                if (modal.resetConfirmOpen) layers.push('reset-confirm');
                if (layers.length) text += ' + ' + layers.join(', ');
            }

            screenLine.textContent = text;
        }

        const data = collectStorage();
        lastFullDump = JSON.stringify(data.full, null, 2);

        if (jsonEl) jsonEl.textContent = JSON.stringify(data.preview, null, 2);
        if (metaEl) metaEl.textContent = 'КЛЮЧЕЙ: ' + data.count + ' · ~' + data.kb + ' КБ';
    }

    function copyText(text) {
        if (navigator.clipboard && typeof navigator.clipboard.writeText === 'function') {
            return navigator.clipboard.writeText(text).then(function () {
                return true;
            }).catch(function () {
                return legacyCopy(text);
            });
        }
        return Promise.resolve(legacyCopy(text));
    }

    function legacyCopy(text) {
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.top = '-1000px';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        let ok = false;
        try { ok = document.execCommand('copy'); } catch (e) { ok = false; }
        ta.remove();
        return ok;
    }

    function copyStorage() {
        syncInfo();
        copyText(lastFullDump).then(function (ok) {
            flash(ok ? 'LOCALSTORAGE СКОПИРОВАН' : 'НЕ УДАЛОСЬ СКОПИРОВАТЬ');
        });
    }

        } catch (e) { /* ignore */ }
    }

    function setToggle(btn, on) {

        const state = btn.querySelector('[data-state]');

    function syncToggles() {
        if (!stage) return;

        const loader = loaderMod();

        const debugOn = loader ? !!loader.DEBUG_SIMULATE_ERROR : store.get(DEBUG_KEY) === 'true';
        const maintOn = loader ? !!loader.MAINTENANCE_MODE : store.get(MAINT_KEY) === 'true';

        setToggle(stage.querySelector('[data-action="toggle-debug"]'), debugOn);
        setToggle(stage.querySelector('[data-action="toggle-maint"]'), maintOn);
        setToggle(stage.querySelector('[data-action="toggle-verbose"]'), verbose);
    }

    function setDisabled(selector, disabled) {
        const btn = stage.querySelector(selector);
        if (btn) btn.disabled = !!disabled;
    }

    function syncActions() {
        if (!stage) return;
        setDisabled('[data-action="skip-loading"]', !isScreenActive('loader-screen'));
        setDisabled('[data-action="skip-intro"]', !isScreenActive('intro-screen'));
    }

    function syncAll() {
        syncToggles();
        syncActions();
        syncInfo();
        setConfirm(false);
    }

    function onKeyDown(e) {
        if (!isOpen) return;

        if (e.key === 'Escape') {
            e.preventDefault();
            e.stopImmediatePropagation();
            close();
        }
    }

    function checkHash() {
        if (window.location.hash.toLowerCase().indexOf('admin') !== -1) {
            if (!isOpen) open();
        }
    }

    // ---- ИНИЦИАЛИЗАЦИЯ --------------------------------------
    function init() {
        if (initialized) return;
        initialized = true;

        verbose = store.get(VERBOSE_KEY) === 'true';

        // Авто-вход по admins.js (до проверки #admin)
        tryLocalUnlock();

        document.addEventListener('keydown', onKeyDown, true);
        window.addEventListener('hashchange', checkHash);

        checkHash();
        log('init');
    }

    if (document.readyState === 'loading') {
        window.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }

    return {
        open: open,
        close: close,
        init: init
    };
})();

// YouTube-style ambient: фон, который живёт вместе с экраном и подхватывает цвет видео
const Ambient = {
    stage: null,
    _reduced: false,
    _active: null,
    _timer: null,
    _generation: 0,

    init() {
        this._reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

        // На узких экранах ambient-слои не нужны — экономим батарею и GPU
        const wide = window.matchMedia('(min-width: 900px)');
        if (wide.matches) this._buildStage();
        // Окно раздвинули до десктопного размера — включаем слои без перезагрузки
        wide.addEventListener('change', (e) => {
            if (e.matches) this._buildStage();
        });
    },

    // Медленно дрейфующие пятна света за игровым экраном (видны в полях по бокам на ПК)
    _buildStage() {
        if (this.stage) return;

        const stage = document.createElement('div');
        stage.className = 'ambient-stage';
        stage.setAttribute('aria-hidden', 'true');
        stage.innerHTML = '<i></i><i></i><i></i>';
        document.body.appendChild(stage);
        this.stage = stage;

        requestAnimationFrame(() => stage.classList.add('ready'));
    },

    /**
     * Следит за видео и перерисовывает его кадры в маленький canvas.
     * CSS размывает этот canvas — получается свечение в цвет ролика,
     * как ambient mode у видео на YouTube.
     *
     * @param {HTMLVideoElement} video
     * @param {string|HTMLCanvasElement} canvasTarget
     * @param {{ interval?: number }} options
     */
    followVideo(video, canvasTarget, options = {}) {
        this.stop();
        if (!video) return;

        const canvas = typeof canvasTarget === 'string'
            ? document.querySelector(canvasTarget)
            : canvasTarget;
        if (!canvas || !canvas.getContext) return;

        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        canvas.classList.add('is-live');

        const gen = ++this._generation;
        // При «меньше движения» обновляем цвет реже, чтобы не мельтешило
        const interval = this._reduced ? 1600 : Math.max(120, options.interval || 320);

        const draw = () => {
            if (gen !== this._generation) return;

            if (video.readyState >= 2 && video.videoWidth > 0) {
                // Квадратный кроп по центру — как object-fit: cover у кружка
                const size = Math.min(video.videoWidth, video.videoHeight);
                const sx = (video.videoWidth - size) / 2;
                const sy = (video.videoHeight - size) / 2;
                try {
                    ctx.drawImage(video, sx, sy, size, size, 0, 0, canvas.width, canvas.height);
                } catch (e) {
                    // Кадр ещё не готов — просто дождёмся следующего
                }
            }

            // В фоновой вкладке не тратим ресурсы на частые перерисовки
            this._timer = setTimeout(draw, document.hidden ? interval * 4 : interval);
        };

        this._active = { video, canvas };
        draw();
    },

    stop() {
        this._generation++;
        if (this._timer) clearTimeout(this._timer);
        this._timer = null;
        if (this._active && this._active.canvas) {
            this._active.canvas.classList.remove('is-live');
        }
        this._active = null;
    }
};

window.addEventListener('DOMContentLoaded', () => Ambient.init());

const Intro = {
    skip: false,
    isTyping: false,
    currentInterval: null,
    typingCompleteCallback: null,
    clickHandler: null,
    finished: false,

    // Скорость автобуса (синхронизирована с анимацией driveRightToLeft в intro.css)
    BUS_DRIVE_MS: 14000,
    BUS_SPRITE_MS: 180,

    async init() {
        // Сброс всех настроек при каждом новом запуске
        this.skip = false;
        this.isTyping = false;
        this.finished = false;
        this.removeClickHandler();
        if (this.currentInterval) clearInterval(this.currentInterval);

        document.querySelectorAll('.intro-scene').forEach(el => el.classList.remove('active'));
        document.getElementById('intro-text').style.opacity = '1';
        document.getElementById('intro-text').textContent = '';
        document.getElementById('intro-text').classList.remove('is-typing', 'waiting');
        document.getElementById('dialogue-text').textContent = '';
        document.getElementById('dialogue-text').classList.remove('is-typing');
        document.getElementById('dialogue-text').style.opacity = '1';
        document.getElementById('dialogue-role').textContent = '';
        document.getElementById('dialogue-role').removeAttribute('data-who');
        document.getElementById('dialogue-next').classList.remove('visible');

        document.getElementById('phone-incoming').classList.remove('show');
        document.getElementById('phone-accept').classList.remove('show');
        document.getElementById('bus-img').classList.remove('drive');
        document.getElementById('bus-img').src = "assets/images/intro/bus-1.png";

        // Кнопка пропуска
        const skipBtn = document.getElementById('btn-skip-intro');
        const newSkipBtn = skipBtn.cloneNode(true); // Убираем старые обработчики
        skipBtn.parentNode.replaceChild(newSkipBtn, skipBtn);
        newSkipBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            this.skip = true;
            this.finish();
        });

        await this.play();
    },

    sleep(ms) {
        return new Promise(resolve => setTimeout(resolve, this.skip ? 0 : ms));
    },

    typeText(text, elementId = 'intro-text', speed = 40) {
        return new Promise(resolve => {
            if (this.currentInterval) clearInterval(this.currentInterval);

            const el = document.getElementById(elementId);
            el.textContent = '';

            if (this.skip) {
                el.textContent = text;
                el.classList.remove('is-typing');
                resolve();
                return;
            }

            this.isTyping = true;
            el.classList.add('is-typing');
            this.typingCompleteCallback = resolve;
            let i = 0;

            this.finishTyping = () => {
                if (!this.isTyping) return;
                clearInterval(this.currentInterval);
                this.isTyping = false;
                el.classList.remove('is-typing');
                el.textContent = text;
                if (this.typingCompleteCallback) {
                    this.typingCompleteCallback();
                    this.typingCompleteCallback = null;
                }
            };

            this.currentInterval = setInterval(() => {
                if (i < text.length) {
                    i++;
                    // textContent безопасен и не ломает теги
                    el.textContent = text.substring(0, i);
                } else {
                    this.finishTyping();
                }
            }, speed);
        });
    },

    waitForClick() {
        return new Promise(resolve => {
            // Скип уже нажат — не ждём клика и не вешаем обработчик
            if (this.skip) { resolve(); return; }

            const screen = document.getElementById('intro-screen');
            const introText = document.getElementById('intro-text');

            // На всякий случай снимаем предыдущий обработчик, чтобы не копить их
            this.removeClickHandler();

            // В диалоге уже есть своя стрелочка на панели, подсказка нужна только рассказчику
            const dialogueActive = document.getElementById('scene-dialogue').classList.contains('active');
            if (!dialogueActive) introText.classList.add('waiting');

            const proceed = () => {
                introText.classList.remove('waiting');
                resolve();
            };

            this.clickHandler = (e) => {
                if (e.target.closest('#btn-skip-intro')) return;

                if (this.isTyping) {
                    this.finishTyping();
                    return; // Клик завершил печать, ждем следующий клик для продолжения
                }

                this.removeClickHandler();
                proceed();
            };
            screen.addEventListener('click', this.clickHandler);
        });
    },

    removeClickHandler() {
        if (!this.clickHandler) return;
        const screen = document.getElementById('intro-screen');
        if (screen) screen.removeEventListener('click', this.clickHandler);
        this.clickHandler = null;
    },

    showScene(sceneId) {
        document.querySelectorAll('.intro-scene').forEach(el => el.classList.remove('active'));
        const scene = document.getElementById(sceneId);
        if (scene) scene.classList.add('active');
    },

    async play() {
        if (this.skip) return this.finish();

        // 1. ОФИС
        this.showScene('scene-office');
        await this.sleep(1000);
        await this.typeText("Вторник, 9:47.\nКофе остыл ещё в семь.\nЕщё один день,\nкоторый никто не запомнит.");
        await this.waitForClick();
        if (this.skip) return this.finish();

        // 2. ЗВОНОК
        this.showScene('scene-phone');
        await this.sleep(500);
        document.getElementById('phone-incoming').classList.add('show');

        await this.typeText("Среди уведомлений —\nодно без фото и без скидки.\n\n«Продам ферму. Недорого.\nСамовывоз.»\n\nНи адреса. Только номер.");
        await this.waitForClick();
        if (this.skip) return this.finish();

        document.getElementById('phone-incoming').classList.remove('show');
        document.getElementById('phone-accept').classList.add('show');
        await this.sleep(300);

        await this.typeText("Тот самый номер.\nТы отвечаешь раньше,\nчем успеваешь придумать, зачем.");
        await this.waitForClick();
        if (this.skip) return this.finish();

        // 3. ДИАЛОГ
        document.getElementById('intro-text').style.opacity = '0';
        this.showScene('scene-dialogue');
        const nextArrow = document.getElementById('dialogue-next');
        const roleEl = document.getElementById('dialogue-role');
        const textEl = document.getElementById('dialogue-text');

        await this.sleep(500);

        const dialog = [
            { role: "ДЕД", who: "ded", text: "Алло? Внучок? Это дед." },
            { role: "ТЫ", who: "you", text: "Дед... ты правда продаёшь ферму?" },
            { role: "ДЕД", who: "ded", text: "Не продаю. Отдаю.\nПриезжай, пока я не передумал." },
            { role: "ТЫ", who: "you", text: "Почему сейчас?" },
            { role: "ДЕД", who: "ded", text: "Дом стоит. Земля ждёт.\nА я устал ждать один." },
            { role: "ТЫ", who: "you", text: "Я же ничего в этом не понимаю." },
            { role: "ДЕД", who: "ded", text: "Научишься. Остальное — на месте.\nЕсть только один нюанс." },
            { role: "ТЫ", who: "you", text: "Какой?" },
            { role: "ДЕД", who: "ded", text: "Енот. Он не мой.\nНо ты его не обижай.\nЛадно, автобус ждёт." }
        ];

        for (let i = 0; i < dialog.length; i++) {
            const line = dialog[i];
            nextArrow.classList.remove('visible');

            // Мягкая смена реплики: старая гаснет, новая появляется
            textEl.style.opacity = '0';
            await this.sleep(140);
            roleEl.textContent = line.role;
            roleEl.dataset.who = line.who;
            textEl.style.opacity = '1';

            await this.typeText(line.text, 'dialogue-text', 30);

            // Даем глазу привыкнуть к тексту (0.4 сек)
            await this.sleep(400);

            nextArrow.classList.add('visible');
            await this.waitForClick();

            if (this.skip) return this.finish();
        }

        nextArrow.classList.remove('visible');
        // 4. АВТОБУС
        const introTextEl = document.getElementById('intro-text');
        introTextEl.textContent = ''; 
        introTextEl.style.opacity = '1'; 

        this.showScene('scene-bus');
        await this.sleep(500);

        await this.typeText("Неделя.\nОдин чемодан.\nИ ни одного плана.");

        const busImg = document.getElementById('bus-img');
        busImg.classList.add('drive');

        const busInterval = setInterval(() => {
            if (this.skip) return;
            busImg.src = busImg.src.includes('bus-1')
                ? "assets/images/intro/bus-2.png"
                : "assets/images/intro/bus-1.png";
        }, this.BUS_SPRITE_MS);

        // Ждём почти до конца поездки автобуса (14 секунд)
        await this.sleep(this.BUS_DRIVE_MS - 800);

        // 🎯 ПЛАВНОЕ ЗАТУХАНИЕ: текст гаснет пока автобус ещё доезжает
        introTextEl.style.opacity = '0';
        await this.sleep(1000); // Даём transition (0.6s) полностью отработать

        clearInterval(busInterval);
        if (this.skip) return this.finish();

        // 5. ДОМ
        this.showScene('scene-house');
        await this.sleep(900); // Даём сцене дома плавно проявиться (0.9s transition)

        // 🎯 ПЛАВНОЕ ПОЯВЛЕНИЕ: новый текст проявляется вместе с началом печати
        introTextEl.textContent = '';
        introTextEl.style.opacity = '1';
        await this.sleep(100); // Микро-пауза, чтобы opacity-transition начался

        await this.typeText("Асфальт позади.\nПоле впереди.\nТишина такая, что слышно,\nкак ты дышишь.\n\nПочти.");

        await this.waitForClick();
        this.finish();
    },

    finish() {
        if (this.finished) return;
        this.finished = true;
        this.skip = true;
        this.removeClickHandler();

        if (this.currentInterval) {
            clearInterval(this.currentInterval);
            this.currentInterval = null;
        }

        // Убираем курсор печати и подсказку, если скип застал текст на середине
        document.querySelectorAll('#intro-screen .is-typing').forEach(el => el.classList.remove('is-typing'));
        document.getElementById('intro-text').classList.remove('waiting');

        // Разблокируем play(), если печать ещё шла в момент скипа
        if (this.typingCompleteCallback) {
            const done = this.typingCompleteCallback;
            this.typingCompleteCallback = null;
            this.isTyping = false;
            done();
        }

        // Интро пройдено — игра считается начатой.
        // При следующем заходе в меню будет «Продолжить» вместо «Играть».
        Progress.start();

        const intro = document.getElementById('intro-screen');
        intro.style.transition = 'opacity 1s ease';
        intro.style.opacity = '0';

        setTimeout(() => {
            intro.classList.add('hidden');
            intro.style.opacity = '1'; // Возвращаем на место для следующего раза
            ScreenManager.showScreen('game-started-screen');
        }, 1000);
    }
};

        // Останавливаем вращение советов, если оно запускалось
        if (this.hintInterval) clearInterval(this.hintInterval);
        if (this.progressInterval) clearInterval(this.progressInterval);

        if (this.maintenanceMessage && this.MAINTENANCE_REASON) {
            this.maintenanceMessage.textContent = this.MAINTENANCE_REASON;
        }

        // Задаем путь к видео из настроек
        const videoElem = document.getElementById('maintenance-video');
        if (videoElem && this.MAINTENANCE_VIDEO) {
            videoElem.src = this.MAINTENANCE_VIDEO;
            // Свечение вокруг кружка подхватывает цвет кадров ролика
            if (typeof Ambient !== 'undefined') Ambient.followVideo(videoElem, '#maintenance-ambient');
        }

        this.buildMaintenanceScene();

        if (this.maintenanceBox) {
            this.maintenanceBox.classList.remove('hidden-element');
        }
    },

    hideMaintenance() {
        const box = this.maintenanceBox || document.getElementById('loader-maintenance');
        if (box) box.classList.add('hidden-element');
    },

    // Звёзды и светлячки для атмосферного фона техработ
    _maintenanceSceneBuilt: false,
    buildMaintenanceScene() {
        if (this._maintenanceSceneBuilt) return;

        const starsBox = document.getElementById('maintenance-stars');
        const firefliesBox = document.getElementById('maintenance-fireflies');
        if (!starsBox && !firefliesBox) return;

        const rand = (min, max) => min + Math.random() * (max - min);

        if (starsBox) {
            const fragment = document.createDocumentFragment();
            for (let i = 0; i < 64; i++) {
                const star = document.createElement('span');
                star.className = 'mstar' + (Math.random() > 0.88 ? ' mstar--big' : '');
                star.style.setProperty('--x', rand(0, 100).toFixed(2) + '%');
                // Больше звёзд в верхней части неба

                star.style.setProperty('--s', rand(1.2, 2.6).toFixed(1) + 'px');
                star.style.setProperty('--dur', rand(2.4, 6.5).toFixed(2) + 's');
                star.style.setProperty('--delay', rand(0, 6).toFixed(2) + 's');
                fragment.appendChild(star);

        if (firefliesBox) {

                fly.className = 'mfirefly';
                fly.style.setProperty('--x', rand(4, 96).toFixed(2) + '%');
                fly.style.setProperty('--y', rand(62, 94).toFixed(2) + '%');
                fly.style.setProperty('--s', rand(2, 4).toFixed(1) + 'px');
                fly.style.setProperty('--dur', rand(6, 12).toFixed(2) + 's');
                fly.style.setProperty('--delay', rand(0, 8).toFixed(2) + 's');
                fragment.appendChild(fly);
            }
            firefliesBox.appendChild(fragment);
        }

        this._maintenanceSceneBuilt = true;
    },

    hints: [

    hintInterval: null,

    // Флаги из скрытой админ-панели: читаем до проверки режима техработ
    applyAdminFlags() {
        try {
            const savedDebug = localStorage.getItem('admin_DEBUG_ERROR');
            const savedMaint = localStorage.getItem('admin_MAINTENANCE');
            if (savedDebug !== null) this.DEBUG_SIMULATE_ERROR = savedDebug === 'true';
            if (savedMaint !== null) this.MAINTENANCE_MODE = savedMaint === 'true';
        } catch (e) {
            // localStorage может быть недоступен — работаем со значениями по умолчанию
        }

    init() {

        this.maintenanceBox = document.getElementById('loader-maintenance');

        // Если включен режим техработ — блокируем всё и останавливаем запуск
        if (this.MAINTENANCE_MODE) {

        this.progressBar = document.getElementById('progress-bar');
        this.statusText = document.getElementById('loader-status');

        this.errorBox = document.getElementById('loader-error');

        this.loaderGif = document.getElementById('loader-gif');

        this.btnContinue = document.getElementById('btn-continue');
        this.btnWait = document.getElementById('btn-wait');
        this.btnWarningRetry = document.getElementById('btn-warning-retry');

        if (this.btnRetry) {

    },

    startHintLoop() {
        let lastIndex = Math.floor(Math.random() * this.hints.length);
        this.showNextHint(lastIndex);
        
        if (this.hintInterval) clearInterval(this.hintInterval);
        this.hintInterval = setInterval(() => {
            let newIndex;
            do {
                newIndex = Math.floor(Math.random() * this.hints.length);
            } while (newIndex === lastIndex && this.hints.length > 1);
            
            lastIndex = newIndex;
            this.showNextHint(newIndex);
        }, 3500);
    },

    showNextHint(index) {
        if (this.hintText) {
            this.hintText.textContent = this.hints[index];
        }
    },

    // Собираем плоский список файлов: { src, type }
    buildAssetList() {
        const images = (this.assetsToLoad.images || []).map(src => ({ src, type: 'image' }));
        const audio = (this.assetsToLoad.audio || []).map(src => ({ src, type: 'audio' }));
        return images.concat(audio);
    },

    // assets — необязательный список: при повторе грузим только то, что упало
    startLoadingProcess(assets) {
        const list = Array.isArray(assets) && assets.length ? assets : this.buildAssetList();

        // Номер попытки: колбэки от прошлого запуска больше не трогают экран
        const run = ++this.runId;

        this.startTime = Date.now();
        this.hasError = false;
        this.failedAssets = [];
        this.loadedCount = 0;
        this.totalCount = list.length || 1;

        // При повторе не заставляем ждать весь брендовый таймер заново
        const isRetryRun = Array.isArray(assets) && assets.length > 0;
        this.runMinTime = isRetryRun ? Math.min(this.minLoadTime, 1200) : this.minLoadTime;

        this.resetLoaderUI();
        this.progressInterval = setInterval(() => this.updateProgress(), 50);

        // Жёлтый флаг, если загрузка затянулась
        this.scheduleSlowWarning(run, this.warningDelay);

                // === ТОЧКА СИМУЛЯЦИИ ОШИБКИ ===
        if (this.DEBUG_SIMULATE_ERROR && !this.hadRetry) {
            setTimeout(() => {
                if (run !== this.runId) return;
                // Убрали push фейкового файла, чтобы при повторе грузились реальные
                this.handleNetworkError(run);
            }, this.DEBUG_ERROR_DELAY);
            return;
        }

        if (!list.length) {
            this.finishLoading(run);
            return;
        }

        Promise.all(list.map(asset => this.loadAsset(asset, run))).then(() => {
            if (run !== this.runId || this.hasError) return;
            this.waitForMinimumTime(run);
        });
    },

    resetLoaderUI() {
        if (this.progressInterval) clearInterval(this.progressInterval);
        if (this.completionTimer) clearInterval(this.completionTimer);
        this.completionTimer = null;
        this.clearSlowWarningTimer();

        if (this.warningBox) this.warningBox.classList.add('hidden-element');
        if (this.errorBox) this.errorBox.classList.add('hidden-element');
        if (this.btnRetry) {
            this.btnRetry.classList.add('hidden-element');
            this.btnRetry.textContent = '↻ Повторить попытку';
        }
        if (this.btnContinue) this.btnContinue.classList.add('hidden-element');
        if (this.successBox) this.successBox.classList.add('hidden-element');
        if (this.loaderGif) this.loaderGif.classList.remove('dimmed');
        if (this.progressBar) {
            this.progressBar.style.width = '0%';
            this.progressBar.classList.remove('is-stalled', 'is-error');
        }
        if (this.statusText) {
            this.statusText.classList.remove('long-wait', 'is-error');
            this.statusText.style.color = '';
        }
        this.setStatus('Загрузка ресурсов... 0%');
    },

    setStatus(text, tone) {
        if (!this.statusText) return;
        this.statusText.textContent = text;
        this.statusText.classList.remove('long-wait', 'is-error');
        if (tone === 'slow') this.statusText.classList.add('long-wait');
        if (tone === 'error') this.statusText.classList.add('is-error');
    },

    // Один файл: всегда завершается — успехом, ошибкой или таймаутом
    loadAsset(asset, run) {
        return new Promise((resolve) => {
            if (run !== this.runId) { resolve(); return; }

            let settled = false;
            const finish = (ok) => {
                if (settled) return;
                settled = true;
                clearTimeout(timeoutId);

                if (run !== this.runId) { resolve(); return; }

                if (ok) {
                    this.loadedCount++;
                } else {
                    this.failedAssets.push(asset);
                    this.handleNetworkError(run);
                }
                resolve();
            };

            // Файл не должен держать загрузку вечно
            const timeoutId = setTimeout(() => finish(false), this.assetTimeout);

            if (asset.type === 'audio') {
                const audio = new Audio();
                audio.preload = 'auto';
                audio.oncanplaythrough = () => finish(true);
                // Пропавший звук не должен блокировать игру
                audio.onerror = () => finish(true);
                audio.src = asset.src;
                audio.load();
                return;
            }

            const img = new Image();
            img.onload = () => finish(true);
            img.onerror = () => finish(false);
            img.src = asset.src;
        });
    },

    // Даём прогрессу доиграть до 5 секунд, но не после ошибки
    waitForMinimumTime(run) {
        if (this.completionTimer) clearInterval(this.completionTimer);

        this.completionTimer = setInterval(() => {
            if (run !== this.runId) {
                clearInterval(this.completionTimer);
                this.completionTimer = null;
                return;
            }
            if (this.hasError) {
                clearInterval(this.completionTimer);
                this.completionTimer = null;
                return;
            }
            if (Date.now() - this.startTime < this.runMinTime) return;

            clearInterval(this.completionTimer);
            this.completionTimer = null;
            this.finishLoading(run);
        }, 100);
    },

    finishLoading(run) {
        if (run !== this.runId) return;

        this.clearSlowWarningTimer();
        if (this.progressInterval) clearInterval(this.progressInterval);

        if (this.warningBox) this.warningBox.classList.add('hidden-element');
        if (this.progressBar) {
            this.progressBar.classList.remove('is-stalled');
            this.progressBar.style.width = '100%';
        }
        this.setStatus('Загрузка завершена!');

        if (this.hadRetry) {
            this.showSuccessAndProceed();
        } else if (this.btnContinue) {
            this.btnContinue.classList.remove('hidden-element');
        }
    },

    handleNetworkError(run) {
        if (this.hasError || run !== this.runId) return;
        this.hasError = true;

        if (this.progressInterval) clearInterval(this.progressInterval);
        if (this.completionTimer) clearInterval(this.completionTimer);
        this.completionTimer = null;
        this.clearSlowWarningTimer();

        const names = this.failedAssets.map(asset => this.prettyAssetName(asset.src));

                if (this.errorText) {
            if (names.length === 1) {
                this.errorText.textContent = `Файл «${names[0]}» не загрузился. Проверь соединение и попробуй снова.`;
            } else if (names.length > 1) {
                const preview = names.slice(0, 3).join(', ') + (names.length > 3 ? '…' : '');
                this.errorText.textContent = `Не загрузилось файлов: ${names.length} (${preview}). Проверь соединение и попробуй снова.`;
            } else {
                // Специально для симуляции ошибки (когда упавших файлов нет)
                this.errorText.textContent = `Енот перегрыз интернет-кабель! Проверь соединение и попробуй снова.`;
            }
        }

        if (this.warningBox) this.warningBox.classList.add('hidden-element');
        if (this.errorBox) this.errorBox.classList.remove('hidden-element');
        if (this.progressBar) {
            this.progressBar.classList.remove('is-stalled');
            this.progressBar.classList.add('is-error');
        }
        if (this.loaderGif) this.loaderGif.classList.add('dimmed');
        if (this.btnContinue) this.btnContinue.classList.add('hidden-element');

        this.setStatus('Ошибка загрузки', 'error');

        if (this.btnRetry) {
            this.btnRetry.textContent = names.length > 1
                ? `↻ Повторить (${names.length})`
                : '↻ Повторить попытку';
            this.btnRetry.classList.remove('hidden-element');
        }
    },

    prettyAssetName(src) {
        return String(src).split('/').pop() || String(src);
    },

    scheduleSlowWarning(run, delay) {
        this.clearSlowWarningTimer();

        this.warningTimer = setTimeout(() => {
            if (run !== this.runId || this.hasError) return;
            this.showSlowWarning();
        }, delay);
    },

    clearSlowWarningTimer() {
        if (this.warningTimer) clearTimeout(this.warningTimer);
        this.warningTimer = null;
    },

    // Жёлтый флажок: грузится долго, но не сломалось
    showSlowWarning() {
        if (this.hasError) return;

        if (this.warningText) {
            const total = this.totalCount;
            const loaded = Math.min(this.loadedCount, total);
            this.warningText.textContent = total > 1
                ? `Загружено ${loaded} из ${total}. Соединение медленнее обычного — файлы всё ещё идут.`
                : 'Соединение медленнее обычного — файлы всё ещё идут.';
        }

        if (this.warningBox) this.warningBox.classList.remove('hidden-element');
        if (this.progressBar) this.progressBar.classList.add('is-stalled');
        if (this.statusText) this.statusText.classList.add('long-wait');
    },

    dismissSlowWarning() {
        if (this.warningBox) this.warningBox.classList.add('hidden-element');
        if (this.progressBar) this.progressBar.classList.remove('is-stalled');
        if (this.statusText) this.statusText.classList.remove('long-wait');

        // Если так и не догрузится — напомним ещё раз
        this.scheduleSlowWarning(this.runId, this.warningRepeat);
    },

    retryLoading() {
        this.hadRetry = true;

        // Повторяем только упавшие файлы — так быстрее всего
        const failed = this.failedAssets.slice();
        this.startLoadingProcess(failed.length ? failed : null);
    },

    showSuccessAndProceed() {
        clearInterval(this.hintInterval);
        if (this.progressInterval) clearInterval(this.progressInterval);
        if (this.completionTimer) clearInterval(this.completionTimer);
        this.completionTimer = null;
        this.clearSlowWarningTimer();

        // Пока играет галочка — тихо подтягиваем ресурсы интро
        this.warmIntroAssets();

        if (this.loaderGif) this.loaderGif.classList.add('dimmed');
        if (this.warningBox) this.warningBox.classList.add('hidden-element');
        if (this.errorBox) this.errorBox.classList.add('hidden-element');
        if (this.btnRetry) this.btnRetry.classList.add('hidden-element');
        if (this.btnContinue) this.btnContinue.classList.add('hidden-element');
        if (this.progressBar) this.progressBar.classList.remove('is-stalled', 'is-error');
        if (this.successBox) this.successBox.classList.remove('hidden-element');
        this.setStatus("Загрузка завершена!");

        this.playSuccessSound();

        setTimeout(() => {
            this.onComplete();
        }, 1500);
    },

    warmIntroAssets() {
        // Игрок уже проходил интро — эти картинки ему в этой сессии не понадобятся
        if (typeof Progress !== 'undefined' && Progress.hasStarted()) return;

        this.introAssets.forEach(src => {
            const img = new Image();
            img.decoding = 'async';
            img.src = src;
        });
    },

    playSuccessSound() {
        try {
            const audio = new Audio('assets/audio/sfx/success.mp3');
            audio.volume = 0.6;
            
            // Воспроизводим звук
            const playPromise = audio.play();
            
            if (playPromise !== undefined) {
                playPromise.catch(error => {
                    console.log("Браузер заблокировал автозвук до первого клика пользователя на странице:", error);
                });
            }
        } catch (e) {
            console.log("Ошибка воспроизведения звука:", e);
        }
    },

    updateProgress() {
        if (this.hasError) return;

        // 1. Реальный процент загрузки файлов (от 0 до 100)
        const realPercent = this.totalCount > 0 ? (this.loadedCount / this.totalCount) * 100 : 0;

        // 2. Временной процент (сколько % от 5 секунд уже прошло)
        const elapsedTime = Date.now() - this.startTime;
        const timePercent = Math.min(100, (elapsedTime / this.runMinTime) * 100);

        // Итоговый процент — это МИНИМУМ из реального и временного
        // Шкала никогда не опередит ни реальную загрузку, ни таймер в 5 секунд
        const displayPercent = Math.floor(Math.min(realPercent, timePercent));

        if (this.progressBar) this.progressBar.style.width = displayPercent + '%';
        if (this.statusText) {
            this.statusText.textContent = `Загрузка ресурсов... ${displayPercent}%`;
        }
    },

    // Освобождаем DOM загрузчика, когда он больше не нужен.
    // Экран техработ оставляем в DOM (скрытым) для админ-панели.
    disposed: false,
    dispose() {
        if (this.disposed) return;
        this.disposed = true;

        if (this.hintInterval) clearInterval(this.hintInterval);
        if (this.progressInterval) clearInterval(this.progressInterval);
        if (this.completionTimer) clearInterval(this.completionTimer);
        this.clearSlowWarningTimer();
        this.hintInterval = null;
        this.progressInterval = null;
        this.completionTimer = null;

        const screen = document.getElementById('loader-screen');
        if (screen) screen.remove();

        // Экран техработ не удаляем: админка должна уметь открыть его
        // в любой момент, даже после старта игры. Держим скрытым.
        if (this.maintenanceBox) this.maintenanceBox.classList.add('hidden-element');

        this.progressBar = null;
        this.statusText = null;
        this.hintText = null;
        this.warningBox = null;
        this.warningText = null;
        this.successBox = null;
        this.loaderGif = null;
        this.btnRetry = null;
        this.btnContinue = null;
        this.errorBox = null;
        this.errorText = null;
        this.btnWait = null;
        this.btnWarningRetry = null;
    },

    onComplete() {
    const curtain = document.getElementById('transition-curtain');

    if (!curtain) {
        ScreenManager.showScreen('main-menu');
        return;
    }

    // 1. Занавес плавно закрывает экран
    curtain.classList.add('active');

    // 2. Через 550мс (пока занавес полностью чёрный) — переключаем экран
    setTimeout(() => {
        ScreenManager.showScreen('main-menu');

        // 3. Через 300мс (menu уже готов показаться) — открываем занавес
        setTimeout(() => {
            curtain.classList.remove('active');
        }, 300);
    }, 550);
}
};

// Менеджер экранов
const ScreenManager = {
    THEME_CLASSES: ['theme-loader', 'theme-menu', 'theme-intro', 'theme-game', 'theme-credits', 'theme-help'],

    showScreen(screenId) {
        // 1. Скрываем все экраны
        document.querySelectorAll('.screen').forEach(s => {
            s.classList.remove('active');
            s.classList.add('hidden');
        });
        
        // 2. Показываем нужный
        const target = document.getElementById(screenId);
        if (target) {
            target.classList.remove('hidden');
            void target.offsetWidth;
            target.classList.add('active');
        }

        // 3. Меняем тему фона, снимая только классы тем (остальные классы body не трогаем)
        document.body.classList.remove(...this.THEME_CLASSES);
        if (screenId === 'loader-screen') document.body.classList.add('theme-loader');
        else if (screenId === 'main-menu') document.body.classList.add('theme-menu');
        else if (screenId === 'intro-screen') document.body.classList.add('theme-intro');
        else if (screenId === 'game-screen') document.body.classList.add('theme-game');
        else if (screenId === 'game-started-screen') document.body.classList.add('theme-menu');

        if (screenId === 'game-screen' || screenId === 'game-started-screen') {
            this.releaseBootUI();
        } else if (screenId === 'main-menu' && typeof Modal !== 'undefined') {
            // Список помощи готовим в простое, когда меню уже на экране
            Modal.scheduleHelpRender();
        }
    },

    // После старта игры меню/лоадер в этой сессии не используются — освобождаем память
    releaseBootUI() {
        if (typeof Modal !== 'undefined' && typeof Modal.releaseForGame === 'function') {
            Modal.releaseForGame();
        }
        if (typeof Loader !== 'undefined' && typeof Loader.dispose === 'function') {
            Loader.dispose();
        }
    },
    
    // Отдельный метод для оверлеев (Титры, Помощь)
    setAmbientTheme(themeName) {
        document.body.classList.remove(...this.THEME_CLASSES);
        document.body.classList.add('theme-' + themeName);
    }
};

    filterHelp(rawQuery) {
        this.ensureHelpRendered();

        const query = String(rawQuery || '').trim().toLowerCase();
        const items = this.helpItems;
        let visible = 0;

        for (let i = 0; i < items.length; i++) {
            const matches = !query || this.helpIndex[i].includes(query);
            const item = items[i];
            item.classList.toggle('is-hidden', !matches);

            if (matches) {
                visible++;
            } else if (item.classList.contains('open')) {
                item.classList.remove('open');
                const btn = item.querySelector('.faq-question');
                if (btn) btn.setAttribute('aria-expanded', 'false');
            }
        }

        const empty = document.getElementById('help-empty');
        if (empty) empty.classList.toggle('hidden-element', visible > 0);

        this.updateHelpCount(visible);
        this.clearHelpToTopTimer();
        this.setHelpToTopVisible(this.helpScrollEl ? this.helpScrollEl.scrollTop > 320 : false);
    },

                // 2. Еще через 1 сек (чтобы текст успел прочитаться) запускаем енота
                const t2 = setTimeout(() => {
                    const reveal = () => {
                        // Принудительный рефлоу, чтобы анимация началась с нуля
                        void flyingRaccoon.offsetWidth;
                        flyingRaccoon.classList.add('animate');

                        // Свечение вокруг кружка подстраивается под цвет ролика
                        if (typeof Ambient !== 'undefined') Ambient.followVideo(creditsVideo, '#credits-ambient');
                    };

                    // Снимаем display:none сразу: ролик ещё невидим (opacity: 0),
                    // но браузер успевает отрисовать первый кадр
                    flyingRaccoon.classList.remove('hidden-element');

                    if (!creditsVideo) {
                        reveal();
                        return;
                    }

                    // Если ролик уже крутится (повторный вход) — показываем сразу
                    if (!creditsVideo.paused && creditsVideo.readyState >= 2) {
                        reveal();
                        return;
                    }

                    // Ждём первый реальный кадр видео, иначе енот появляется
                    // резко на чёрном кружке, пока грузится ролик
                    let guard = null;
                    const show = () => {
                        creditsVideo.removeEventListener('playing', show);
                        if (guard) clearTimeout(guard);
                        reveal();
                    };
                    creditsVideo.addEventListener('playing', show);
                    creditsVideo.play().catch(() => {});
                    // Страховка, если браузер не запустит видео сам
                    guard = setTimeout(show, 1500);
                    this.creditsTimers.push(guard);
                }, 1000);
                this.creditsTimers.push(t2);

                // Дальше экран не закрывается сам — ждём клика по финальной карточке
            }, 2500);
            this.creditsTimers.push(t1);
        };
        this.creditsAnimationHandler = onScrollEnd;
        scroll.addEventListener('animationend', onScrollEnd, { once: true });
    },

    closeCredits() {
        ScreenManager.setAmbientTheme('menu');
        if (typeof Ambient !== 'undefined') Ambient.stop();
        const screen = document.getElementById('credits-screen');
        if (!screen) return;
        this.clearCreditsTimers();
        screen.classList.remove('active');
        setTimeout(() => {
            screen.classList.add('hidden-element');
            const thanks = screen.querySelector('.credits-thanks');
            if (thanks) thanks.classList.remove('visible');
            this.creditsOpen = false;
        }, 800);
    },

    // === ОСВОБОЖДЕНИЕ ПАМЯТИ ПОСЛЕ СТАРТА ИГРЫ ===
    // FAQ-разметка и звёзды нужны только меню и титрам. В игре не держим их в DOM.
    releaseForGame() {
        this.helpRenderToken++;
        if (this.helpSearchTimer) {
            clearTimeout(this.helpSearchTimer);
            this.helpSearchTimer = null;
        }
        this.clearHelpToTopTimer();
        this.clearCreditsTimers();

        if (this.helpRendered) {
            const list = document.getElementById('faq-list');
            if (list) list.replaceChildren();
            this.helpItems = [];
            this.helpIndex = [];
            this.helpRendered = false;
        }

        document.querySelectorAll('.credits-stars').forEach(box => box.replaceChildren());
    },

    clearCreditsTimers() {
        this.creditsTimers.forEach(t => clearTimeout(t));
        this.creditsTimers = [];

        // Снимаем слушатель, иначе он сработает от следующей анимации
        if (this.creditsAnimationHandler) {
            const scroll = document.querySelector('#credits-screen .credits-scroll');
            if (scroll) scroll.removeEventListener('animationend', this.creditsAnimationHandler);
            this.creditsAnimationHandler = null;
        }
    }
};

// === ПРОГРЕСС ИГРЫ ===
// Пока реальной игры нет, прогрессом считается пройденное интро.
// Флаг хранится в localStorage и переживает перезагрузку страницы.
const Progress = {
    KEY: 'amberfields_save',

    load() {
        try {
            const raw = localStorage.getItem(this.KEY);
            if (!raw) return null;
            const data = JSON.parse(raw);
            return data && typeof data === 'object' ? data : null;
        } catch (e) {
            return null;
        }
    },

    save(data) {
        try {
            localStorage.setItem(this.KEY, JSON.stringify(data));
        } catch (e) {
            // localStorage может быть недоступен (приватный режим) — не ломаем игру
        }
    },

    hasStarted() {
        const data = this.load();
        return !!(data && data.started);
    },

    // Игрок начал игру: запоминаем это для следующего захода
    start() {
        const data = this.load() || {};
        data.started = true;
        data.startedAt = data.startedAt || Date.now();
        data.lastPlayed = Date.now();
        this.save(data);
    },

    clear() {
        try {
            localStorage.removeItem(this.KEY);
        } catch (e) {
            // см. save()
        }
    }
};

const Viewport = {
    GAME_WIDTH: 480,
    GAME_HEIGHT: 854,
    MAX_SCALE_PC: 1.5,
    ASPECT_THRESHOLD: 0.25,

    init() {
        this.update();
        window.addEventListener('resize', () => this.update());
        window.addEventListener('orientationchange', () => {
            setTimeout(() => this.update(), 150);
        });
    },

    update() {
    const vw = window.innerWidth;
    const vh = window.innerHeight;

    const gameAspect = this.GAME_WIDTH / this.GAME_HEIGHT;
    const screenAspect = vw / vh;
    const aspectDiff = Math.abs(screenAspect - gameAspect) / gameAspect;

    let transform;
    let stretchRatio = 1;   // во сколько раз экран растянут по вертикали относительно горизонтали

    if (aspectDiff < this.ASPECT_THRESHOLD) {
        // ТЕЛЕФОН: растягиваем неравномерно
        const scaleX = vw / this.GAME_WIDTH;
        const scaleY = vh / this.GAME_HEIGHT;
        stretchRatio = scaleY / scaleX;
        transform = `translate(-50%, -50%) scale(${scaleX}, ${scaleY})`;
    } else {
        // ПК: uniform scale
        let scale = Math.min(vw / this.GAME_WIDTH, vh / this.GAME_HEIGHT);
        if (scale > this.MAX_SCALE_PC) scale = this.MAX_SCALE_PC;
        transform = `translate(-50%, -50%) scale(${scale})`;
    }

    // Сохраняем коэффициент растяжения для CSS
    document.documentElement.style.setProperty('--stretch-ratio', stretchRatio);

    document.querySelectorAll('.screen').forEach(el => {
        el.style.transform = transform;
    });
    }
};

window.addEventListener('DOMContentLoaded', () => Viewport.init());
